## 1. 头文件

```c
#include "ch32v30x_i2c.h"   // 文件: Peripheral/inc/ch32v30x_i2c.h
```

## 2. 数据结构

```c
typedef struct {
    uint32_t I2C_ClockSpeed;          // 时钟频率 (<400kHz)
    uint16_t I2C_Mode;                // @I2C_mode
    uint16_t I2C_DutyCycle;           // 占空比 (快速模式)
    uint16_t I2C_OwnAddress1;         // 自身地址
    uint16_t I2C_Ack;                 // 应答
    uint16_t I2C_AcknowledgedAddress; // 地址模式 (7/10位)
} I2C_InitTypeDef;
```

## 3. 关键宏

```c
// 模式
#define I2C_Mode_I2C           ((uint16_t)0x0000)
#define I2C_Mode_SMBusDevice   ((uint16_t)0x0002)
#define I2C_Mode_SMBusHost     ((uint16_t)0x000A)

// 占空比 (快速模式 400kHz)
#define I2C_DutyCycle_16_9    ((uint16_t)0x4000)  // t_low/t_high = 16/9
#define I2C_DutyCycle_2       ((uint16_t)0xBFFF)  // t_low/t_high = 2

// 应答
#define I2C_Ack_Enable        ((uint16_t)0x0400)
#define I2C_Ack_Disable       ((uint16_t)0x0000)

// 地址模式
#define I2C_AcknowledgedAddress_7bit  ((uint16_t)0x4000)
#define I2C_AcknowledgedAddress_10bit ((uint16_t)0xC000)

// 方向
#define I2C_Direction_Transmitter ((uint8_t)0x00)
#define I2C_Direction_Receiver    ((uint8_t)0x01)

// ════════════════════════════════════════════
// 事件标志 (主机模式)
// ════════════════════════════════════════════
#define I2C_EVENT_MASTER_MODE_SELECT             ((uint32_t)0x00030001)  // 起始条件已发送
#define I2C_EVENT_MASTER_TRANSMITTER_MODE_SELECTED ((uint32_t)0x00070082) // 地址已发送(写)
#define I2C_EVENT_MASTER_RECEIVER_MODE_SELECTED  ((uint32_t)0x00030002)  // 地址已发送(读)
#define I2C_EVENT_MASTER_BYTE_TRANSMITTING       ((uint32_t)0x00070080)  // 正在发送数据
#define I2C_EVENT_MASTER_BYTE_TRANSMITTED        ((uint32_t)0x00070084)  // 数据已发送(带ACK)
#define I2C_EVENT_MASTER_BYTE_RECEIVED           ((uint32_t)0x00030040)  // 接收到数据
```

## 6. 典型用法

### 初始化

```c
void I2C1_Init(void) {
    I2C_InitTypeDef I2C_InitStructure = {0};
    GPIO_InitTypeDef GPIO_InitStructure = {0};

    RCC_APB1PeriphClockCmd(RCC_APB1Periph_I2C1, ENABLE);
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOB, ENABLE);

    // PB6 SCL, PB7 SDA (复用开漏)
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_6 | GPIO_Pin_7;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_AF_OD;
    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_Init(GPIOB, &GPIO_InitStructure);

    I2C_InitStructure.I2C_ClockSpeed = 100000;         // 100KHz
    I2C_InitStructure.I2C_Mode = I2C_Mode_I2C;
    I2C_InitStructure.I2C_DutyCycle = I2C_DutyCycle_2;
    I2C_InitStructure.I2C_OwnAddress1 = 0x00;
    I2C_InitStructure.I2C_Ack = I2C_Ack_Enable;
    I2C_InitStructure.I2C_AcknowledgedAddress = I2C_AcknowledgedAddress_7bit;
    I2C_Init(I2C1, &I2C_InitStructure);
    I2C_Cmd(I2C1, ENABLE);
}
```

### 主机发送数据

```c
void I2C_Master_Transmit(uint8_t slave_addr, uint8_t* data, uint8_t len) {
    while (I2C_GetFlagStatus(I2C1, I2C_FLAG_BUSY));

    I2C_GenerateSTART(I2C1, ENABLE);
    while (!I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_MODE_SELECT));

    I2C_Send7bitAddress(I2C1, slave_addr, I2C_Direction_Transmitter);
    while (!I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_TRANSMITTER_MODE_SELECTED));

    for (uint8_t i = 0; i < len; i++) {
        I2C_SendData(I2C1, data[i]);
        while (!I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_BYTE_TRANSMITTED));
    }

    I2C_GenerateSTOP(I2C1, ENABLE);
}
```

### 主机接收数据

```c
void I2C_Master_Receive(uint8_t slave_addr, uint8_t* buf, uint8_t len) {
    while (I2C_GetFlagStatus(I2C1, I2C_FLAG_BUSY));

    I2C_GenerateSTART(I2C1, ENABLE);
    while (!I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_MODE_SELECT));

    I2C_Send7bitAddress(I2C1, slave_addr, I2C_Direction_Receiver);
    while (!I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_RECEIVER_MODE_SELECTED));

    for (uint8_t i = 0; i < len; i++) {
        if (i == len - 1) I2C_AcknowledgeConfig(I2C1, DISABLE);  // 最后一字节NACK
        while (!I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_BYTE_RECEIVED));
        buf[i] = I2C_ReceiveData(I2C1);
    }

    I2C_GenerateSTOP(I2C1, ENABLE);
    I2C_AcknowledgeConfig(I2C1, ENABLE);
}
```
