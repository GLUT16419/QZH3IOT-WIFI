# 其他外设模块

包含：PWR(电源管理)、RTC(实时时钟)、IWDG(看门狗)、BKP(备份寄存器)、CRC、RNG(随机数)、FPU(浮点运算)、FLASH操作。

---

## 1. PWR 电源管理

```c
#include "ch32v30x_pwr.h"   // 文件: Peripheral/inc/ch32v30x_pwr.h

void PWR_DeInit(void);
void PWR_PVDCmd(FunctionalState NewState);
void PWR_PVDLevelConfig(uint32_t PWR_PVDLevel);
void PWR_WakeUpPinCmd(FunctionalState NewState);
void PWR_EnterSTOPMode(uint32_t PWR_Regulator, uint8_t PWR_STOPEntry);
void PWR_EnterSTANDBYMode(void);

#define PWR_PVDLevel_2V2  ((uint32_t)0x00000000)  // ~2.9V
#define PWR_PVDLevel_2V9  ((uint32_t)0x000000E0)
#define PWR_Regulator_ON  ((uint32_t)0x00000000)
#define PWR_Regulator_LowPower ((uint32_t)0x00000100)
#define PWR_STOPEntry_WFI  ((uint8_t)0x01)
#define PWR_STOPEntry_WFE  ((uint8_t)0x02)
```

## 3. IWDG 独立看门狗

```c
#include "ch32v30x_iwdg.h"   // 文件: Peripheral/inc/ch32v30x_iwdg.h

void IWDG_WriteAccessCmd(uint16_t IWDG_WriteAccess);
void IWDG_SetPrescaler(uint8_t IWDG_Prescaler);
void IWDG_SetReload(uint16_t Reload);
void IWDG_ReloadCounter(void);   // 喂狗
void IWDG_Enable(void);          // 启动看门狗

#define IWDG_Prescaler_4    ((uint8_t)0x00)   // LSI/4
#define IWDG_Prescaler_8    ((uint8_t)0x01)
#define IWDG_Prescaler_16   ((uint8_t)0x02)
#define IWDG_Prescaler_32   ((uint8_t)0x03)
#define IWDG_Prescaler_64   ((uint8_t)0x04)
#define IWDG_Prescaler_128  ((uint8_t)0x05)
#define IWDG_Prescaler_256  ((uint8_t)0x06)

// 典型配置: 预分频256, 重载1000 -> 约1秒超时 (LSI≈40KHz)
void IWDG_Init(void) {
    IWDG_WriteAccessCmd(IWDG_WriteAccess_Enable);
    IWDG_SetPrescaler(IWDG_Prescaler_256);
    IWDG_SetReload(1000);
    IWDG_ReloadCounter();
    IWDG_Enable();
}
// 循环中: IWDG_ReloadCounter();  // 喂狗
```

## 7. FPU 浮点运算单元

```c
// CH32V307内置硬件单精度FPU
// 在main函数前使能:
#define FPU_ENABLE  (*(volatile uint32_t*)0xE000ED88) |= (1<<20)

// 然后直接使用float运算即可:
// float a = 3.14f, b = 2.0f;
// float c = a * b + 1.0f;  // 由硬件完成

// 注意: double会被编译器转换为软件模拟，性能较低
// 建议全部使用 float 类型
```

## 8. FLASH 闪存操作

```c
#include "ch32v30x_flash.h"   // 文件: Peripheral/inc/ch32v30x_flash.h

// 状态枚举
typedef enum {
    FLASH_BUSY = 1,
    FLASH_ERROR_PG,
    FLASH_ERROR_WRP,
    FLASH_COMPLETE,
    FLASH_TIMEOUT
} FLASH_Status;

// 函数原型
void FLASH_DeInit(void);
void FLASH_Unlock(void);                    // 解锁
void FLASH_Lock(void);                      // 锁定
FLASH_Status FLASH_ErasePage(uint32_t Page_Address);       // 擦除页 (1KB/页)
FLASH_Status FLASH_EraseAllPages(void);                    // 擦除全部
FLASH_Status FLASH_ProgramWord(uint32_t Address, uint32_t Data);        // 写32位
FLASH_Status FLASH_ProgramHalfWord(uint32_t Address, uint16_t Data);    // 写16位
FLASH_Status FLASH_ProgramOptionByteData(uint32_t Address, uint8_t Data);
uint32_t FLASH_GetOptionByte(uint32_t Address);
uint32_t FLASH_GetReadOutProtectionStatus(void);

// 典型用法:
// FLASH_Unlock();
// FLASH_ErasePage(0x08010000);           // 擦除第64页
// FLASH_ProgramWord(0x08010000, 0x12345678);
// FLASH_Lock();
```
