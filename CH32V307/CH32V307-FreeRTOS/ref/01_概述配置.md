# FreeRTOS 概述与配置模块

包含：项目概述、目录结构、FreeRTOSConfig.h 完整配置、基础数据类型、中断控制宏。

---

## 1. 项目概述

- **芯片**: CH32V307VCT6 (RISC-V RV32IMAC, 144MHz)
- **RTOS**: FreeRTOS V10.4.6 (移植)
- **内存管理**: heap_4.c
- **Tick频率**: 500Hz (2ms/Tick)，使用 TIM2 硬件定时器
- **调度**: 抢占式

## 3. FreeRTOSConfig.h 完整配置

必须按此配置，不能修改关键参数（否则可能与现有移植不兼容）：

```c
#ifndef FREERTOS_CONFIG_H
#define FREERTOS_CONFIG_H

#include "ch32v30x.h"   // 提供 SystemCoreClock

// ── RISC-V 芯片特定 ──
#define configMTIME_BASE_ADDRESS      0     // CH32V307无mtime，必须为0
#define configMTIMECMP_BASE_ADDRESS   0     // CH32V307无mtimecmp，必须为0

// ── 调度器 ──
#define configUSE_PREEMPTION          1     // 抢占式调度
#define configUSE_IDLE_HOOK           0
#define configUSE_TICK_HOOK           0
#define configUSE_TICKLESS_IDLE       0

// ── 时钟 ──
#define configCPU_CLOCK_HZ            SystemCoreClock  // 144MHz
#define configTICK_RATE_HZ            ((TickType_t)500) // 500Hz = 2ms

// ── 任务与堆 ──
#define configMAX_PRIORITIES          (15)
#define configMINIMAL_STACK_SIZE      ((unsigned short)256)   // 字
#define configTOTAL_HEAP_SIZE         ((size_t)(12 * 1024))   // 12KB
#define configMAX_TASK_NAME_LEN       (16)
#define configUSE_16_BIT_TICKS        0

// ── 功能 ──
#define configUSE_MUTEXES             1
#define configUSE_RECURSIVE_MUTEXES   1
#define configUSE_COUNTING_SEMAPHORES 1
#define configUSE_TIMERS              1
#define configUSE_CO_ROUTINES         0
#define configUSE_QUEUE_SETS          0
#define configUSE_TASK_NOTIFICATIONS  1

// ── 定时器任务 ──
#define configTIMER_TASK_PRIORITY     (configMAX_PRIORITIES - 1)
#define configTIMER_QUEUE_LENGTH      10
#define configTIMER_TASK_STACK_DEPTH  (configMINIMAL_STACK_SIZE * 2)

// ── 检测 ──
#define configCHECK_FOR_STACK_OVERFLOW         1
#define configQUEUE_REGISTRY_SIZE              10

// ── API可用性 ──
#define INCLUDE_vTaskPrioritySet                1
#define INCLUDE_uxTaskPriorityGet               1
#define INCLUDE_vTaskDelete                     1
#define INCLUDE_vTaskSuspend                    1
#define INCLUDE_vTaskDelayUntil                 1
#define INCLUDE_vTaskDelay                      1
#define INCLUDE_xTaskGetSchedulerState          1
#define INCLUDE_xTaskGetCurrentTaskHandle       1
#define INCLUDE_uxTaskGetStackHighWaterMark     1
#define INCLUDE_xEventGroupSetBitFromISR        1

// ── 中断优先级 ──
#define configPRIO_BITS                2
#define configKERNEL_INTERRUPT_PRIORITY 0xFF
#define configMAX_API_CALL_INTERRUPT_PRIORITY   0

#endif
```

## 4. 基础数据类型

```c
// 定义在 portmacro.h
typedef uint32_t    StackType_t;      // 栈帧类型 (4字节)
typedef int32_t     BaseType_t;       // 基础类型
typedef uint32_t    TickType_t;       // Tick计数 (32位)

// 返回值
#define pdTRUE      ((BaseType_t)1)
#define pdFALSE     ((BaseType_t)0)
#define pdPASS      (pdTRUE)
#define pdFAIL      (pdFALSE)
#define portMAX_DELAY           (TickType_t)0xFFFFFFFFUL   // 无限等待
#define portNO_DELAY            (TickType_t)0

// Tick转换
#define pdMS_TO_TICKS(xTimeMs)  ((TickType_t)(((uint32_t)(xTimeMs)) / portTICK_PERIOD_MS))
#define portTICK_PERIOD_MS      ((TickType_t)2)            // 每个Tick=2ms
```

## 5. 中断控制宏

```c
// 关闭中断（清除mstatus的MIE位）
#define portDISABLE_INTERRUPTS()   __asm volatile("csrc mstatus, 8")
// 开启中断（设置mstatus的MIE位）
#define portENABLE_INTERRUPTS()    __asm volatile("csrs mstatus, 8")
// 中断屏蔽 (RISC-V特性，直接返回0/无操作)
#define portSET_INTERRUPT_MASK_FROM_ISR()    0
#define portCLEAR_INTERRUPT_MASK_FROM_ISR(x)  (void)x
```
