# FreeRTOS 移植层细节、常见问题与完整项目模板

## 1. 移植层关键细节

### 上下文切换流程

```c
// 文件: FreeRTOS/portable/GCC/RISC-V/port.c + portASM.S

// 上下文切换由以下方式触发:
// 1. portYIELD()      -> 触发 ecall 软件中断
// 2. xPortSysTickHandler() -> Tick中断 (TIM2)
// 3. portYIELD_FROM_ISR()  -> ISR中请求切换

// 上下文切换保存的寄存器:
// ra (返回地址), sp (栈指针), gp (全局指针), tp (线程指针)
// s0-s11 (Callee-saved寄存器)

// 注意: CH32V307 不支持标准RISC-V的 mtime/mtimecmp
// 因此 configMTIME_BASE_ADDRESS 和 configMTIMECMP_BASE_ADDRESS 必须为0
```

### 启动文件中的中断向量

```c
// 文件: Startup/startup_ch32v30x_D8C.S

// 使用 .weak 声明，允许用户重写:
.weak NMI_Handler
.weak HardFault_Handler
.weak TIM2_IRQHandler     // FreeRTOS Tick
.weak SysTick_Handler     // 备用Tick
// ...
```

### 编译选项

## 2. 常见问题

### Q1: 任务创建失败 (返回NULL)

```c
// 原因1: 堆内存不足
// 解决: 增大 configTOTAL_HEAP_SIZE 或减小栈大小

// 原因2: 栈大小太小 (至少256字)
// 解决: 确保 usStackDepth >= configMINIMAL_STACK_SIZE

// 调试: 使用 uxTaskGetNumberOfTasks() 查看已创建任务数
```

### Q2: HardFault / 死机

```c
// 原因1: 栈溢出
// 解决:
//   - 实现 vApplicationStackOverflowHook
//   - 增大任务栈
//   - 使用 uxTaskGetStackHighWaterMark 监控

// 原因2: 中断嵌套导致栈溢出
// 解决: 确保 ISR 中不使用过多局部变量

// 原因3: 在 ISR 中调用了非 FromISR 版本的 API
// 解决: 使用 xQueueSendFromISR 替代 xQueueSend 等
```

### Q3: vTaskDelay 不精确

```c
// 原因: configTICK_RATE_HZ = 500, 每个Tick = 2ms
// vTaskDelay(1) 实际延时 2ms (不是1ms)

// 需要精确延时 (<2ms) 时:
//   使用普通延时 Delay_Us() / Delay_Ms()

// 需要精确周期时:
//   使用 vTaskDelayUntil()
```

### Q4: printf 输出乱序/混乱

```c
// 原因: 多个任务同时调用 printf 不是线程安全的
// 解决: 使用互斥锁保护

void vSafePrintf(const char *fmt, ...) {
    xSemaphoreTake(xPrintfMutex, portMAX_DELAY);
    printf(...);
    xSemaphoreGive(xPrintfMutex);
}
```

### Q5: WCHNET + FreeRTOS 共用 TIM2

```c
// 两者都可能使用 TIM2，冲突时:
// 方案1: 将 WCHNET 改为其他定时器 (如 TIM3)
// 方案2: 合并两个中断，在一个 TIM2_IRQHandler 中同时处理:
void TIM2_IRQHandler(void) {
    if (TIM_GetITStatus(TIM2, TIM_IT_Update) != RESET) {
        TIM_ClearITPendingBit(TIM2, TIM_IT_Update);
        WCHNET_PeriodicHandle();     // 协议栈定时 (50ms)
        if (xTaskGetSchedulerState() != taskSCHEDULER_NOT_STARTED) {
            xPortSysTickHandler();   // FreeRTOS Tick (2ms)
        }
    }
}
// 注意: 需要将 WCHNET 的 TIM2 预分频调整为 2ms 周期
// 并在 WCHNET_PeriodicHandle() 内部做 50ms 计数判断
```

## 3. 完整项目模板

```c
// main.c - FreeRTOS + CH32V307 完整项目模板

#include "debug.h"
#include "FreeRTOS.h"
#include "task.h"
#include "queue.h"
#include "semphr.h"
#include "timers.h"
#include "event_groups.h"

// ── 任务参数 ──
#define TASK_A_PRIO     5
#define TASK_A_STK      256
#define TASK_B_PRIO     4
#define TASK_B_STK      256

TaskHandle_t xTaskAHandle;
TaskHandle_t xTaskBHandle;

// ── 栈溢出钩子 ──
void vApplicationStackOverflowHook(TaskHandle_t xTask, char *pcTaskName) {
    printf("Stack Overflow! Task: %s\r\n", pcTaskName);
    while(1);
}

// ── GPIO初始化 ──
static void prvGPIO_Init(void) {
    GPIO_InitTypeDef GPIO_InitStructure = {0};
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOA, ENABLE);
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_0 | GPIO_Pin_1;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_Out_PP;
    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_Init(GPIOA, &GPIO_InitStructure);
}

// ── 任务实现 ──
static void prvTaskA(void *pvParameters) {
    while(1) {
        GPIO_SetBits(GPIOA, GPIO_Pin_0);
        vTaskDelay(pdMS_TO_TICKS(500));
        GPIO_ResetBits(GPIOA, GPIO_Pin_0);
        vTaskDelay(pdMS_TO_TICKS(500));
    }
}

static void prvTaskB(void *pvParameters) {
    TickType_t xLastWakeTime = xTaskGetTickCount();
    while(1) {
        GPIO_Toggle(GPIOA, GPIO_Pin_1);
        vTaskDelayUntil(&xLastWakeTime, pdMS_TO_TICKS(1000));
    }
}

// ── 主函数 ──
int main(void) {
    NVIC_PriorityGroupConfig(NVIC_PriorityGroup_2);
    SystemCoreClockUpdate();
    Delay_Init();
    USART_Printf_Init(115200);

    printf("FreeRTOS CH32V307 Template\r\n");
    printf("SysClk: %d Hz\r\n", SystemCoreClock);

    prvGPIO_Init();

    xTaskCreate(prvTaskA, "TaskA", TASK_A_STK, NULL, TASK_A_PRIO, &xTaskAHandle);
    xTaskCreate(prvTaskB, "TaskB", TASK_B_STK, NULL, TASK_B_PRIO, &xTaskBHandle);

    vTaskStartScheduler();  // 启动调度器

    while(1);  // 不会执行到此处
}
```

### TIM2 Tick 初始化 (加入 FreeRTOS)

```c
// 放在 main() 中, vTaskStartScheduler 之前
void vTIM2_Tick_Init(void) {
    TIM_TimeBaseInitTypeDef TIM_TimeBaseStructure = {0};
    RCC_APB1PeriphClockCmd(RCC_APB1Periph_TIM2, ENABLE);
    TIM_TimeBaseStructure.TIM_Prescaler = SystemCoreClock / 1000000 - 1;
    TIM_TimeBaseStructure.TIM_Period = 2000 - 1;   // 2ms
    TIM_TimeBaseStructure.TIM_CounterMode = TIM_CounterMode_Up;
    TIM_TimeBaseStructure.TIM_ClockDivision = 0;
    TIM_TimeBaseInit(TIM2, &TIM_TimeBaseStructure);
    TIM_ITConfig(TIM2, TIM_IT_Update, ENABLE);
    TIM_Cmd(TIM2, ENABLE);
    NVIC_EnableIRQ(TIM2_IRQn);
}
```
