# FreeRTOS 任务管理模块

包含：任务创建/删除/延时/挂起/恢复/控制完整API。

---

## 1. 头文件

```c
#include "task.h"       // FreeRTOS/include/task.h
#include "FreeRTOS.h"   // 必须在前
```

## 2. 创建任务

```c
BaseType_t xTaskCreate(
    TaskFunction_t   pvTaskCode,         // void task(void *param)
    const char * const pcName,           // 任务名 (调试用)
    const uint16_t   usStackDepth,       // 栈大小 (字! 不是字节)
    void * const     pvParameters,       // 参数指针
    UBaseType_t      uxPriority,         // 优先级 0~14
    TaskHandle_t * const pxCreatedTask   // [OUT] 句柄 (可NULL)
);
// 返回: pdPASS 成功, errCOULD_NOT_ALLOCATE_REQUIRED_MEMORY 失败

// 静态创建 (栈由调用者提供):
BaseType_t xTaskCreateStatic(
    TaskFunction_t pvTaskCode,
    const char * const pcName,
    const uint32_t ulStackDepth,
    void * const pvParameters,
    UBaseType_t uxPriority,
    StackType_t * const puxStackBuffer,   // 预分配栈
    StaticTask_t * const pxTaskBuffer,    // 预分配TCB
    TaskHandle_t * const pxCreatedTask
);
```

**优先级说明**: 0 = 最低, configMAX_PRIORITIES-1 = 14 = 最高

## 3. 删除任务

```c
void vTaskDelete(TaskHandle_t xTask);
// xTask = NULL 表示删除自己
// 删除后不再执行，已分配的内存不会自动释放!
```

## 4. 延时函数

```c
// 相对延时: 从调用开始延时N个Tick
void vTaskDelay(const TickType_t xTicksToDelay);
// 例: vTaskDelay(pdMS_TO_TICKS(100));  // 延时100ms (等于50 Ticks)

// 绝对延时: 保持固定周期运行
// 用法:
TickType_t xLastWakeTime = xTaskGetTickCount();
while(1) {
    vTaskDelayUntil(&xLastWakeTime, pdMS_TO_TICKS(10));  // 精确每10ms一次
    // 任务代码...
}
```

## 5. 挂起/恢复

```c
void vTaskSuspend(TaskHandle_t xTaskToSuspend);         // 挂起
void vTaskResume(TaskHandle_t xTaskToResume);            // 恢复
BaseType_t xTaskResumeFromISR(TaskHandle_t xTaskToResume); // 从中断恢复
```

## 6. 任务控制函数

```c
void vTaskPrioritySet(TaskHandle_t xTask, UBaseType_t uxNewPriority);
UBaseType_t uxTaskPriorityGet(const TaskHandle_t xTask);

TickType_t xTaskGetTickCount(void);         // 当前Tick数
TickType_t xTaskGetTickCountFromISR(void);  // 中断中调用

UBaseType_t uxTaskGetNumberOfTasks(void);               // 任务数量
UBaseType_t uxTaskGetStackHighWaterMark(TaskHandle_t xTask); // 栈剩余(字)

void vTaskStartScheduler(void);   // 启动调度器 (不会返回)
void vTaskEndScheduler(void);     // 停止调度器

BaseType_t xTaskGetSchedulerState(void);
TaskHandle_t xTaskGetCurrentTaskHandle(void);
```

## 7. 完整示例

```c
#include "debug.h"
#include "FreeRTOS.h"
#include "task.h"

TaskHandle_t xTask1Handle, xTask2Handle;

void vTask1(void *pvParameters) {
    while(1) {
        GPIO_SetBits(GPIOA, GPIO_Pin_0);
        vTaskDelay(pdMS_TO_TICKS(500));     // 500ms
        GPIO_ResetBits(GPIOA, GPIO_Pin_0);
        vTaskDelay(pdMS_TO_TICKS(500));
    }
}

void vTask2(void *pvParameters) {
    TickType_t xLastWakeTime = xTaskGetTickCount();
    while(1) {
        GPIO_Toggle(GPIOA, GPIO_Pin_1);
        vTaskDelayUntil(&xLastWakeTime, pdMS_TO_TICKS(1000));  // 严格每1秒
    }
}

int main(void) {
    NVIC_PriorityGroupConfig(NVIC_PriorityGroup_2);
    SystemCoreClockUpdate();
    Delay_Init();
    USART_Printf_Init(115200);

    printf("FreeRTOS Version:%s\r\n", tskKERNEL_VERSION_NUMBER);

    GPIO_Toggle_INIT();

    // 优先级5, 栈256字
    xTaskCreate(vTask1, "Task1", 256, NULL, 5, &xTask1Handle);
    xTaskCreate(vTask2, "Task2", 256, NULL, 5, &xTask2Handle);

    vTaskStartScheduler();  // 启动调度器
    while(1);               // 不会运行到这里
}
```

---

*注意: 任务栈单位是"字" (StackType_t = 4字节)，256字 = 1024字节*
